project(d2r_frame LANGUAGES C CXX RC)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

target_link_libraries(d2rl-vrcam PRIVATE d2rcam minhook user32 gdi32 ${MINGW_ROOT}/x86_64-w64-mingw32/lib/libwinpthread.a)
