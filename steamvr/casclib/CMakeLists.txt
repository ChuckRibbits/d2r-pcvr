cmake_minimum_required(VERSION 3.15)
project(D2RPCVRCascLib LANGUAGES C CXX)
if(NOT CASCLIB_SOURCE_DIR)
  message(FATAL_ERROR "Set CASCLIB_SOURCE_DIR to pinned CascLib 3.0 sources")
endif()
set(CASC_UNICODE ON CACHE BOOL "Unicode installation paths" FORCE)
set(CASC_BUILD_SHARED_LIB ON CACHE BOOL "Installer-only DLL" FORCE)
set(CASC_BUILD_STATIC_LIB OFF CACHE BOOL "" FORCE)
set(CASC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
add_subdirectory("${CASCLIB_SOURCE_DIR}" casclib)
target_link_libraries(casc ws2_32)
set_target_properties(casc PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
